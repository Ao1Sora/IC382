cmake_minimum_required(VERSION 3.8)
project(ic382_planners)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
add_library(ic382_clearance_lattice SHARED src/clearance_lattice.cpp src/staged_goal_controller.cpp)
target_compile_features(ic382_clearance_lattice PUBLIC cxx_std_17)
ament_target_dependencies(ic382_clearance_lattice nav2_core nav2_smac_planner
  nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle tf2_ros nav2_regulated_pure_pursuit_controller)
pluginlib_export_plugin_description_file(nav2_core planners.xml)
install(TARGETS ic382_clearance_lattice LIBRARY DESTINATION lib)
ament_package()
